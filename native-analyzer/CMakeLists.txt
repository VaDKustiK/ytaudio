cmake_minimum_required(VERSION 3.20)
project(ytaudio_analyzer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../build/analyzer/deps" CACHE PATH "Verified dependencies")
set(ORT_ROOT "${DEPS}/ort" CACHE PATH "ONNX Runtime CPU SDK")
add_library(onnxruntime SHARED IMPORTED)
if(WIN32)
  set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/onnxruntime.dll" IMPORTED_IMPLIB "${ORT_ROOT}/lib/onnxruntime.lib")
elseif(APPLE)
  set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.dylib")
else()
  set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.so")
endif()
set_target_properties(onnxruntime PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${ORT_ROOT}/include")
add_executable(ytaudio-analyzer src/main.cpp "${DEPS}/beat/Source/MelSpectrogram.cpp" "${DEPS}/beat/Source/InferenceProcessor.cpp")
target_include_directories(ytaudio-analyzer PRIVATE src "${DEPS}" "${DEPS}/beat/Source" "${DEPS}/pocketfft" "${DEPS}/r8brain")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/analyzer.json" ANALYZER_METADATA)
string(JSON ENGINE_VERSION GET "${ANALYZER_METADATA}" engine_version)
string(JSON MODELS_VERSION GET "${ANALYZER_METADATA}" models_version)
target_compile_definitions(ytaudio-analyzer PRIVATE YTAUDIO_ENGINE_VERSION="${ENGINE_VERSION}" YTAUDIO_MODELS_VERSION="${MODELS_VERSION}")
target_link_libraries(ytaudio-analyzer PRIVATE onnxruntime)
# The upstream inference translation unit uses memcpy without including cstring.
if(MSVC)
  target_compile_options(ytaudio-analyzer PRIVATE /utf-8 /FIcstring)
else()
  target_compile_options(ytaudio-analyzer PRIVATE -include cstring)
  target_link_libraries(ytaudio-analyzer PRIVATE ${CMAKE_DL_LIBS})
endif()
if(APPLE)
  set_target_properties(ytaudio-analyzer PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(ytaudio-analyzer PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
include(CTest)
add_executable(rhythm-test tests/rhythm_test.cpp)
target_include_directories(rhythm-test PRIVATE src)
add_test(NAME rhythm COMMAND rhythm-test)
install(TARGETS ytaudio-analyzer RUNTIME DESTINATION .)
add_executable(harmony-test tests/harmony_test.cpp)
target_include_directories(harmony-test PRIVATE src "${DEPS}/pocketfft")
add_test(NAME harmony COMMAND harmony-test)
